36pub struct WorkspaceSnapshot {
37    /// Note id → NoteObs
38    pub notes: HashMap<NoteId, NoteObs>,
39    /// Checkbox id → CheckboxObs
40    pub checkboxes: HashMap<CheckboxId, CheckboxObs>,
41    /// Note id → all checklist IDs in source order.
42    pub note_checkboxes: HashMap<NoteId, Vec<CheckboxId>>,
43    /// Checkbox id → direct child checkbox IDs in source order.
44    pub checkbox_children: HashMap<CheckboxId, Vec<CheckboxId>>,
45    /// Note id → list of note IDs that reference this note via @ID ref items (backlinks).
46    pub note_backlinks: HashMap<NoteId, Vec<NoteId>>,
47    /// Checkbox id → parent checkbox id (absent for root-level items).
48    pub checkbox_parent: HashMap<CheckboxId, CheckboxId>,
49    /// Config-driven default values for known metadata fields.
50    pub metadata_defaults: HashMap<String, Value>,
51}

53impl WorkspaceSnapshot {
54    pub fn observe_checked(&self, id: &CheckboxId) -> Option<Status> {
55        self.checkboxes.get(id).map(|c| {
56            if c.checked {
57                Status::Done
58            } else {
59                Status::Todo
60            }
61        })
62    }
63
64    /// Generic metadata observation.
65    pub fn observe_meta(&self, note_id: &NoteId, field: &str) -> Value {
66        self.notes
67            .get(note_id)
68            .and_then(|obs| obs.raw_meta.get(field).cloned())
69            .or_else(|| self.metadata_defaults.get(field).cloned())
70            .unwrap_or_else(|| Value::String(Rc::<str>::from("")))
71    }
72
73    pub fn has_metadata_field(&self, note_id: &NoteId, field: &str) -> bool {
74        self.notes
75            .get(note_id)
76            .map(|obs| obs.raw_meta.contains_key(field))
77            .unwrap_or(false)
78            || self.metadata_defaults.contains_key(field)
79    }
80
81    pub fn targets(&self, id: &CheckboxId) -> &[NoteId] {
82        self.checkboxes
83            .get(id)
84            .map(|c| c.targets.as_slice())
85            .unwrap_or(&[])
86    }
87
88    /// All checklist items in source order.
89    pub fn local_checkboxes(&self, note_id: &NoteId) -> &[CheckboxId] {
90        self.note_checkboxes
91            .get(note_id)
92            .map(|v| v.as_slice())
93            .unwrap_or(&[])
94    }
95
96    pub fn children(&self, checkbox_id: &CheckboxId) -> &[CheckboxId] {
97        self.checkbox_children
98            .get(checkbox_id)
99            .map(|v| v.as_slice())
100            .unwrap_or(&[])
101    }
102
103    pub fn backlinks(&self, note_id: &NoteId) -> &[NoteId] {
104        self.note_backlinks
105            .get(note_id)
106            .map(|v| v.as_slice())
107            .unwrap_or(&[])
108    }
109
110    pub fn parent(&self, id: &CheckboxId) -> Option<&CheckboxId> {
111        self.checkbox_parent.get(id)
112    }
113
114    pub fn all_note_ids(&self) -> impl Iterator<Item = &NoteId> {
115        self.notes.keys()
116    }
117
118    pub fn all_checkbox_ids(&self) -> impl Iterator<Item = &CheckboxId> {
119        self.checkboxes.keys()
120    }
121
122    #[allow(dead_code)]
123    pub fn note_obs(&self, note_id: &NoteId) -> Option<&NoteObs> {
124        self.notes.get(note_id)
125    }

144    pub fn from_note_map_with_metadata_records(
145        notes: &HashMap<NoteId, (PathBuf, String)>,
146        metadata_records: &HashMap<NoteId, MetadataRecord>,
147        metadata_fields: &[MetadataFieldConfig],
148    ) -> Self {
149        let mut note_obs_map: HashMap<NoteId, NoteObs> = HashMap::new();
150        let mut checkboxes: HashMap<CheckboxId, CheckboxObs> = HashMap::new();
151        let mut note_checkboxes: HashMap<NoteId, Vec<CheckboxId>> = HashMap::new();
152        let mut checkbox_children: HashMap<CheckboxId, Vec<CheckboxId>> = HashMap::new();
153        let mut checkbox_parent: HashMap<CheckboxId, CheckboxId> = HashMap::new();
154        let metadata_kinds = metadata_kind_map(metadata_fields);
155        let metadata_defaults = metadata_default_map(metadata_fields);
156
157        for (id, (_path, content)) in notes {
158            let (relation, raw_meta) = if let Some(record) = metadata_records.get(id) {
159                let relation = record.relation.clone();
160                let raw_meta = extract_raw_meta_from_record(record, &metadata_kinds);
161                (relation, raw_meta)
162            } else {
163                (Relation::Active, HashMap::new())
164            };
165
166            note_obs_map.insert(id.clone(), NoteObs { relation, raw_meta });
167
168            // Parse checklist items.
169            let items = parser::parse_checklist_items(content);
170            let mut checkbox_ids: Vec<CheckboxId> = Vec::new();
171            let mut stack: Vec<(usize, CheckboxId)> = Vec::new();
172
173            for item in &items {
174                let cid = CheckboxId {
175                    note_id: id.clone(),
176                    line_idx: item.line_idx,
177                };
178
179                let (checked, targets) = match &item.kind {
180                    ChecklistItemKind::Local => (item.checked, vec![]),
181                    ChecklistItemKind::Ref { targets } => {
182                        let ids: Vec<NoteId> =
183                            targets.iter().map(|t| t.target_id.clone()).collect();
184                        (item.checked, ids)
185                    }
186                };
187
188                checkboxes.insert(cid.clone(), CheckboxObs { checked, targets });
189                while stack
190                    .last()
191                    .is_some_and(|(indent, _)| *indent >= item.indent)
192                {
193                    stack.pop();
194                }
195                if let Some((_, parent_id)) = stack.last() {
196                    checkbox_children
197                        .entry(parent_id.clone())
198                        .or_default()
199                        .push(cid.clone());
200                    checkbox_parent.insert(cid.clone(), parent_id.clone());
201                }
202                checkbox_children.entry(cid.clone()).or_default();
203                stack.push((item.indent, cid.clone()));
204                checkbox_ids.push(cid);
205            }
206
207            note_checkboxes.insert(id.clone(), checkbox_ids);
208        }
209
210        // Build reverse backlink index: for each ref item @target, record source → target backlink.
211        let mut note_backlinks: HashMap<NoteId, Vec<NoteId>> =
212            notes.keys().map(|id| (id.clone(), Vec::new())).collect();
213        for (id, cids) in &note_checkboxes {
214            for cid in cids {
215                if let Some(obs) = checkboxes.get(cid) {
216                    for target in &obs.targets {
217                        note_backlinks
218                            .entry(target.clone())
219                            .or_default()
220                            .push(id.clone());
221                    }
222                }
223            }
224        }
225        for backlinks in note_backlinks.values_mut() {
226            backlinks.sort();
227            backlinks.dedup();
228        }
229
230        WorkspaceSnapshot {
231            notes: note_obs_map,
232            checkboxes,
233            note_checkboxes,
234            checkbox_children,
235            note_backlinks,
236            checkbox_parent,
237            metadata_defaults,
238        }
239    }
240}

242fn metadata_kind_map(
243    metadata_fields: &[MetadataFieldConfig],
244) -> HashMap<String, MetadataFieldKind> {
245    metadata_fields
246        .iter()
247        .map(|field| (field.path.clone(), field.kind.clone()))
248        .collect()
249}

251fn metadata_default_map(metadata_fields: &[MetadataFieldConfig]) -> HashMap<String, Value> {
252    let mut defaults: HashMap<String, Value> = metadata_fields
253        .iter()
254        .map(|field| {
255            (
256                field.path.clone(),
257                toml_value_to_typed_value(&field.default, Some(&field.kind)),
258            )
259        })
260        .collect();
261    defaults
262        .entry("checklist-status".to_string())
263        .or_insert(Value::Status(Status::None));
264    defaults
265        .entry("relation".to_string())
266        .or_insert(Value::String(Rc::from("active")));
267    defaults
268}

270fn extract_raw_meta_from_record(
271    record: &MetadataRecord,
272    metadata_kinds: &HashMap<String, MetadataFieldKind>,
273) -> HashMap<String, Value> {
274    let mut table = record.extra.clone();
275    table.insert(
276        "schema-version".into(),
277        toml::Value::Integer(record.schema_version as i64),
278    );
279    table.insert(
280        "aliases".into(),
281        toml::Value::Array(
282            record
283                .aliases
284                .iter()
285                .cloned()
286                .map(toml::Value::String)
287                .collect(),
288        ),
289    );
290    table.insert(
291        "abstract".into(),
292        toml::Value::String(record.abstract_text.clone().unwrap_or_default()),
293    );
294    table.insert(
295        "keywords".into(),
296        toml::Value::Array(
297            record
298                .keywords
299                .iter()
300                .cloned()
301                .map(toml::Value::String)
302                .collect(),
303        ),
304    );
305    table.insert("generated".into(), toml::Value::Boolean(record.generated));
306    table.insert(
307        "checklist-status".into(),
308        toml::Value::String(record.checklist_status.as_str().into()),
309    );
310    table.insert(
311        "relation".into(),
312        toml::Value::String(
313            match record.relation {
314                Relation::Archived => "archived",
315                Relation::Legacy => "legacy",
316                Relation::Active => "active",
317            }
318            .into(),
319        ),
320    );
321    table.insert(
322        "relation-target".into(),
323        toml::Value::Array(
324            record
325                .relation_target
326                .iter()
327                .cloned()
328                .map(toml::Value::String)
329                .collect(),
330        ),
331    );
332
333    let mut raw_meta = HashMap::new();
334    flatten_toml_table("", &table, metadata_kinds, &mut raw_meta);
335    raw_meta.insert(
336        "checklist-status".to_string(),
337        Value::Status(match record.checklist_status {
338            ChecklistStatus::Done => Status::Done,
339            ChecklistStatus::Wip => Status::Wip,
340            ChecklistStatus::Todo => Status::Todo,
341            ChecklistStatus::None => Status::None,
342        }),
343    );
344    raw_meta
345}

347fn flatten_toml_table(
348    prefix: &str,
349    table: &toml::Table,
350    metadata_kinds: &HashMap<String, MetadataFieldKind>,
351    raw_meta: &mut HashMap<String, Value>,
352) {
353    for (key, value) in table {
354        let full_key = if prefix.is_empty() {
355            key.clone()
356        } else {
357            format!("{prefix}.{key}")
358        };
359
360        match value {
361            toml::Value::Table(inner) => {
362                flatten_toml_table(&full_key, inner, metadata_kinds, raw_meta)
363            }
364            _ => {
365                raw_meta.insert(
366                    full_key.clone(),
367                    toml_value_to_typed_value(value, metadata_kinds.get(&full_key)),
368                );
369            }
370        }
371    }
372}
373
374fn toml_value_to_typed_value(value: &toml::Value, kind: Option<&MetadataFieldKind>) -> Value {
375    match kind {
376        Some(MetadataFieldKind::Boolean) => Value::Bool(value.as_bool().unwrap_or(false)),
377        Some(MetadataFieldKind::ArrayString) => Value::List(Rc::new(
378            value
379                .as_array()
380                .map(|items| {
381                    items
382                        .iter()
383                        .filter_map(|item| item.as_str().map(|s| Value::String(Rc::from(s))))
384                        .collect()
385                })
386                .unwrap_or_default(),
387        )),
388        Some(MetadataFieldKind::String) => Value::String(Rc::from(toml_value_to_string(value))),
389        None => Value::String(Rc::from(toml_value_to_string(value))),
390    }
391}
392
393fn toml_value_to_string(value: &toml::Value) -> String {
394    match value {
395        toml::Value::String(s) => s.clone(),
396        toml::Value::Integer(n) => n.to_string(),
397        toml::Value::Float(f) => f.to_string(),
398        toml::Value::Boolean(b) => b.to_string(),
399        toml::Value::Datetime(dt) => dt.to_string(),
400        toml::Value::Array(items) => {
401            let rendered: Vec<String> = items.iter().map(toml_value_to_inline_string).collect();
402            format!("[{}]", rendered.join(", "))
403        }
404        toml::Value::Table(table) => {
405            let mut parts = Vec::new();
406            for (key, item) in table {
407                parts.push(format!("{key} = {}", toml_value_to_inline_string(item)));
408            }
409            format!("{{{}}}", parts.join(", "))
410        }
411    }
412}
413
414fn toml_value_to_inline_string(value: &toml::Value) -> String {
415    match value {
416        toml::Value::String(s) => format!("{s:?}"),
417        _ => toml_value_to_string(value),
418    }
419}

421/// Async constructor for production use.
422#[allow(dead_code)]
423pub async fn build_workspace_snapshot(note_dir: &Path) -> anyhow::Result<WorkspaceSnapshot> {
424    let mut map: HashMap<NoteId, (PathBuf, String)> = HashMap::new();
425    let mut rd = tokio::fs::read_dir(note_dir).await?;
426    while let Some(entry) = rd.next_entry().await? {
427        let path = entry.path();
428        if path.extension().and_then(|e| e.to_str()) != Some("typ") {
429            continue;
430        }
431        let stem = match path.file_stem().and_then(|s| s.to_str()) {
432            Some(s) => s.to_string(),
433            None => continue,
434        };
435        if stem.len() != 10 || !stem.chars().all(|c| c.is_ascii_digit()) {
436            continue;
437        }
438        match tokio::fs::read_to_string(&path).await {
439            Ok(content) => {
440                map.insert(stem, (path, content));
441            }
442            Err(_) => continue,
443        }
444    }
445    Ok(WorkspaceSnapshot::from_note_map(&map))
446}

683        let metadata_fields = vec![
684            MetadataFieldConfig {
685                path: "user.kind".to_string(),
686                kind: MetadataFieldKind::String,
687                default: toml::Value::String(String::new()),
688            },
689            MetadataFieldConfig {
690                path: "user.priority".to_string(),
691                kind: MetadataFieldKind::Boolean,
692                default: toml::Value::Boolean(false),
693            },
694            MetadataFieldConfig {
695                path: "user.tags".to_string(),
696                kind: MetadataFieldKind::ArrayString,
697                default: toml::Value::Array(Vec::new()),
698            },
699        ];
700        let content = make_toml_note_with_extra_metadata("A", "1111111111", "none", "", "");
701        let mut user = toml::Table::new();
702        user.insert("kind".into(), toml::Value::String("project".into()));
703        user.insert("priority".into(), toml::Value::Boolean(true));
704        user.insert(
705            "tags".into(),
706            toml::Value::Array(vec![
707                toml::Value::String("alpha".into()),
708                toml::Value::String("beta".into()),
709            ]),
710        );
711        let extra = toml::Table::from_iter([("user".into(), toml::Value::Table(user))]);
712        let snap = single_note_snapshot_with_record(
713            "1111111111",
714            &content,
715            record(Status::None, Relation::Active, extra),
716            &metadata_fields,
717        );
718
719        assert_eq!(
720            snap.observe_meta(&"1111111111".to_string(), "user.kind"),
721            Value::String("project".into())
722        );
723        assert_eq!(
724            snap.observe_meta(&"1111111111".to_string(), "user.priority"),
725            Value::Bool(true)
726        );
727        assert_eq!(
728            snap.observe_meta(&"1111111111".to_string(), "user.tags"),
729            Value::List(vec![Value::String("alpha".into()), Value::String("beta".into())].into())
730        );
731    }
