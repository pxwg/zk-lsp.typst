56pub async fn run_reconcile(config: &WikiConfig, dry_run: bool) -> Result<ReconcileStats> {
57    let notes = scan_notes(&config.note_dir).await?;
58    let metadata_records = metadata::read_records(config).await?;
59    let (eval_result, diagnostics) = evaluate_workspace(&notes, &metadata_records, config)?;
60
61    if !diagnostics.is_empty() {
62        return Err(anyhow::anyhow!(render_diagnostics(&diagnostics, &notes)));
63    }
64
65    let reconcile_result = materialize(eval_result);
66    let mut files_changed = 0usize;
67    for (_id, (path, content)) in &notes {
68        let checked_by_line: HashMap<usize, CheckboxWriteback> = reconcile_result
69            .materialized_checked
70            .iter()
71            .filter(|(cid, _)| cid.note_id == *_id)
72            .map(|(cid, writeback)| (cid.line_idx, *writeback))
73            .collect();
74
75        let new_content = normalize_note_from_checked(content, &checked_by_line);
76        if new_content != *content {
77            files_changed += 1;
78            if !dry_run {
79                // Atomic write: tmp → rename
80                let tmp = path.with_extension("typ.tmp");
81                tokio::fs::write(&tmp, new_content.as_bytes()).await?;
82                tokio::fs::rename(&tmp, path).await?;
83            } else {
84                eprintln!("  would update: {}", path.display());
85            }
86        }
87    }
88
89    let metadata_patches = materialized_metadata_patches(&reconcile_result);
90    if !metadata_patches.is_empty() {
91        files_changed += 1;
92        if dry_run {
93            eprintln!(
94                "  would update: {}",
95                metadata::metadata_path(&config.root).display()
96            );
97        } else {
98            for (id, patch) in metadata_patches {
99                metadata::patch_record(config, &id, &patch).await?;
100            }
101        }
102    }
103
104    Ok(ReconcileStats { files_changed })
105}

107pub async fn collect_diagnostics(
108    config: &WikiConfig,
109    overlay: Option<(&std::path::Path, &str)>,
110) -> Result<Vec<ReconcileDiagnostic>> {
111    let mut notes = scan_notes(&config.note_dir).await?;
112
113    if let Some((path, content)) = overlay {
114        if path.extension().and_then(|ext| ext.to_str()) == Some("typ") {
115            if let Some(note_id) = path.file_stem().and_then(|stem| stem.to_str()) {
116                if note_id.len() == 10 && note_id.chars().all(|c| c.is_ascii_digit()) {
117                    notes.insert(
118                        note_id.to_string(),
119                        (path.to_path_buf(), content.to_string()),
120                    );
121                }
122            }
123        }
124    }
125
126    let metadata_records = metadata::read_records(config).await.unwrap_or_default();
127    let (_, diagnostics) = evaluate_workspace(&notes, &metadata_records, config)?;
128    Ok(diagnostics)
129}

131fn materialized_metadata_patches(
132    reconcile_result: &materialize::ReconcileResult,
133) -> HashMap<NoteId, HashMap<String, toml::Value>> {
134    let mut patches: HashMap<NoteId, HashMap<String, toml::Value>> = HashMap::new();
135    for ((note_id, field), value) in &reconcile_result.materialized_meta {
136        let Some(toml_value) = value_to_toml(value) else {
137            continue;
138        };
139        patches
140            .entry(note_id.clone())
141            .or_default()
142            .insert(field.clone(), toml_value);
143    }
144    patches
145}

147fn value_to_toml(value: &Value) -> Option<toml::Value> {
148    match value {
149        Value::Bool(value) => Some(toml::Value::Boolean(*value)),
150        Value::Int(n) => Some(toml::Value::Integer(*n)),
151        Value::Nil => None,
152        Value::Status(status) => Some(toml::Value::String(status.to_str().to_string())),
153        Value::String(value) => Some(toml::Value::String(value.to_string())),
154        Value::List(items) => {
155            let mut out = Vec::with_capacity(items.len());
156            for item in items.iter() {
157                out.push(value_to_toml(item)?);
158            }
159            Some(toml::Value::Array(out))
160        }
161        Value::CheckboxWriteback(_) | Value::NoteRef(_) | Value::CheckboxRef(_) => None,
162    }
163}

169async fn scan_notes(note_dir: &std::path::Path) -> Result<HashMap<NoteId, (PathBuf, String)>> {
170    let mut map = HashMap::new();
171    let mut rd = tokio::fs::read_dir(note_dir).await?;
172    while let Some(entry) = rd.next_entry().await? {
173        let path = entry.path();
174        if path.extension().and_then(|e| e.to_str()) != Some("typ") {
175            continue;
176        }
177        let stem = match path.file_stem().and_then(|s| s.to_str()) {
178            Some(s) => s.to_string(),
179            None => continue,
180        };
181        if stem.len() != 10 || !stem.chars().all(|c| c.is_ascii_digit()) {
182            continue;
183        }
184        match tokio::fs::read_to_string(&path).await {
185            Ok(content) => {
186                map.insert(stem, (path, content));
187            }
188            Err(_) => continue,
189        }
190    }
191    Ok(map)
192}

194fn evaluate_workspace(
195    notes: &HashMap<NoteId, (PathBuf, String)>,
196    metadata_records: &HashMap<NoteId, metadata::MetadataRecord>,
197    config: &WikiConfig,
198) -> Result<(eval::EvalResult, Vec<ReconcileDiagnostic>)> {
199    let snapshot = WorkspaceSnapshot::from_note_map_with_metadata_records(
200        notes,
201        metadata_records,
202        &config.zk_config.metadata.fields,
203    );
204    let module = load_module(
205        &config.zk_config.reconcile_rules,
206        config.zk_config.disable_default_reconcile_rules,
207    )?;
208    let type_info =
209        typecheck::type_check_module_with_metadata(&module, &config.zk_config.metadata.fields)
210            .map_err(anyhow::Error::msg)?;
211
212    let eval_result = eval::eval_all_typed(&module, &snapshot, &type_info);
213    let diagnostics = build_diagnostics(notes, eval_result.diagnostics.clone());
214    Ok((eval_result, diagnostics))
215}

217fn build_diagnostics(
218    notes: &HashMap<NoteId, (PathBuf, String)>,
219    eval_diagnostics: Vec<ReconcileDiagnostic>,
220) -> Vec<ReconcileDiagnostic> {
221    let graph = dependency_graph::build_dependency_graph(notes);
222    let mut diagnostics = cycle_diagnostics(&graph);
223    diagnostics.extend(located_eval_diagnostics(eval_diagnostics, notes));
224    diagnostics

227fn cycle_diagnostics(graph: &DependencyGraph) -> Vec<ReconcileDiagnostic> {
228    let cycles = detect_workspace_cycles(graph);
229    let mut diagnostics = Vec::new();
230    for cycle in cycles {
231        let cycle_chain = render_cycle_chain(&cycle.nodes);
232        for edge in &cycle.edges {
233            let related_locations = cycle
234                .edges
235                .iter()
236                .filter(|other| {
237                    !(other.file_path == edge.file_path
238                        && other.line == edge.line
239                        && other.byte_start == edge.byte_start
240                        && other.byte_end == edge.byte_end)
241                })
242                .map(|other| DiagnosticLocation {
243                    file_path: other.file_path.clone(),
244                    line: other.line,
245                    byte_start: other.byte_start,
246                    byte_end: other.byte_end,
247                })
248                .collect();
249            diagnostics.push(ReconcileDiagnostic {
250                note_id: edge.from_note_id.clone(),
251                message: format!("Cyclic task dependency: {cycle_chain}"),
252                kind: DiagnosticKind::Cycle,
253                severity: DiagnosticSeverity::Error,
254                location: Some(DiagnosticLocation {
255                    file_path: edge.file_path.clone(),
256                    line: edge.line,
257                    byte_start: edge.byte_start,
258                    byte_end: edge.byte_end,
259                }),
260                related_locations,
261            });
262        }
263    }
264    diagnostics
265}

267fn located_eval_diagnostics(
268    diagnostics: Vec<ReconcileDiagnostic>,
269    notes: &HashMap<NoteId, (PathBuf, String)>,
270) -> Vec<ReconcileDiagnostic> {
271    diagnostics
272        .into_iter()
273        .map(|mut diag| {
274            if diag.location.is_none() {
275                if let Some((path, _)) = notes.get(&diag.note_id) {
276                    diag.location = Some(DiagnosticLocation {
277                        file_path: path.clone(),
278                        line: 0,
279                        byte_start: 0,
280                        byte_end: 0,
281                    });
282                }
283            }
284            diag
285        })
286        .collect()
287}

289fn detect_workspace_cycles(graph: &DependencyGraph) -> Vec<WorkspaceCycle> {
290    struct TarjanState<'a> {
291        graph: &'a HashMap<String, Vec<String>>,
292        index_counter: usize,
293        stack: Vec<String>,
294        on_stack: HashMap<String, bool>,
295        index: HashMap<String, usize>,
296        lowlink: HashMap<String, usize>,
297        sccs: Vec<Vec<String>>,
298    }
299
300    impl<'a> TarjanState<'a> {
301        fn strongconnect(&mut self, v: &str) {
302            self.index.insert(v.to_string(), self.index_counter);
303            self.lowlink.insert(v.to_string(), self.index_counter);
304            self.index_counter += 1;
305            self.stack.push(v.to_string());
306            self.on_stack.insert(v.to_string(), true);
307
308            if let Some(neighbors) = self.graph.get(v) {
309                let neighbors = neighbors.clone();
310                for w in neighbors {
311                    if !self.index.contains_key(&w) {
312                        self.strongconnect(&w);
313                        let lv = *self.lowlink.get(v).expect("lowlink exists");
314                        let lw = *self.lowlink.get(&w).unwrap_or(&usize::MAX);
315                        self.lowlink.insert(v.to_string(), lv.min(lw));
316                    } else if *self.on_stack.get(&w).unwrap_or(&false) {
317                        let lv = *self.lowlink.get(v).expect("lowlink exists");
318                        let iw = *self.index.get(&w).expect("index exists");
319                        self.lowlink.insert(v.to_string(), lv.min(iw));
320                    }
321                }
322            }
323
324            if self.lowlink.get(v) == self.index.get(v) {
325                let mut scc = Vec::new();
326                loop {
327                    let w = self.stack.pop().expect("stack pop");
328                    self.on_stack.insert(w.clone(), false);
329                    scc.push(w.clone());
330                    if w == v {
331                        break;
332                    }
333                }
334                self.sccs.push(scc);
335            }
336        }
337    }
338
339    let mut nodes = graph.nodes.clone();
340    nodes.sort();
341    let mut state = TarjanState {
342        graph: &graph.adj,
343        index_counter: 0,
344        stack: Vec::new(),
345        on_stack: HashMap::new(),
346        index: HashMap::new(),
347        lowlink: HashMap::new(),
348        sccs: Vec::new(),
349    };
350    for node in &nodes {
351        if !state.index.contains_key(node.as_str()) {
352            state.strongconnect(node);
353        }
354    }
355
356    state
357        .sccs
358        .into_iter()
359        .filter_map(|scc| {
360            let is_self_loop = scc.len() == 1
361                && graph
362                    .adj
363                    .get(&scc[0])
364                    .is_some_and(|neighbors| neighbors.contains(&scc[0]));
365            if scc.len() <= 1 && !is_self_loop {
366                return None;
367            }
368
369            let edges = graph
370                .occurrences
371                .iter()
372                .filter(|occ| scc.contains(&occ.from_note_id) && scc.contains(&occ.to_note_id))
373                .cloned()
374                .collect();
375            Some(WorkspaceCycle { nodes: scc, edges })
376        })
377        .collect()
378}

380fn render_cycle_chain(nodes: &[NoteId]) -> String {
381    let mut chain = nodes.to_vec();
382    chain.sort();
383    if let Some(first) = chain.first().cloned() {
384        chain.push(first);
385    }
386    chain.join(" -> ")
387}

389fn render_diagnostics(
390    diagnostics: &[ReconcileDiagnostic],
391    notes: &HashMap<NoteId, (PathBuf, String)>,
392) -> String {
393    let color = stderr_color_enabled();
394    let mut out = String::new();
395
396    for diag in diagnostics {
397        if color {
398            out.push_str("\x1b[1;31merror\x1b[0m");
399            out.push_str("\x1b[1m: ");
400            out.push_str(&diag.message);
401            out.push_str("\x1b[0m\n");
402        } else {
403            out.push_str("error: ");
404            out.push_str(&diag.message);
405            out.push('\n');
406        }
407
408        let Some(location) = &diag.location else {
409            out.push('\n');
410            continue;
411        };
412
413        let line_1based = location.line + 1;
414        let col_1based = location.byte_start + 1;
415        let line_text = notes
416            .get(&diag.note_id)
417            .and_then(|(_, content)| content.lines().nth(location.line).map(str::to_string))
418            .or_else(|| {
419                std::fs::read_to_string(&location.file_path)
420                    .ok()
421                    .and_then(|content| content.lines().nth(location.line).map(str::to_string))
422            })
423            .unwrap_or_default();
424
425        let lnum_w = format!("{line_1based}").len();
426        let pad = " ".repeat(lnum_w);
427        let prefix_bytes = (location.byte_start as usize).min(line_text.len());
428        let end_bytes = (location.byte_end as usize).min(line_text.len());
429        let display_col = display_width(&line_text[..prefix_bytes]);
430        let underline_disp = display_width(&line_text[prefix_bytes..end_bytes]).max(1);
431        let underline = "^".repeat(underline_disp);
432        let pointer_spaces = " ".repeat(display_col);
433        let path = location.file_path.display();
434
435        if color {
436            out.push_str(&format!(
437                " \x1b[1;34m{pad}┌─\x1b[0m \x1b[36m{path}:{line_1based}:{col_1based}\x1b[0m\n"
438            ));
439            out.push_str(&format!(" \x1b[1;34m{pad}│\x1b[0m\n"));
440            out.push_str(&format!(
441                "\x1b[1;34m{line_1based:>lnum_w$} │\x1b[0m {line_text}\n"
442            ));
443            out.push_str(&format!(
444                " \x1b[1;34m{pad}│\x1b[0m {pointer_spaces}\x1b[1;31m{underline}\x1b[0m\n"
445            ));
446        } else {
447            out.push_str(&format!(" {pad}┌─ {path}:{line_1based}:{col_1based}\n"));
448            out.push_str(&format!(" {pad}│\n"));
449            out.push_str(&format!("{line_1based:>lnum_w$} │ {line_text}\n"));
450            out.push_str(&format!(" {pad}│ {pointer_spaces}{underline}\n"));
451        }
452        out.push('\n');
453    }
454
455    out
456}
