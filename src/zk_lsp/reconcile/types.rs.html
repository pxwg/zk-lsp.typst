1//! Core types for the Reconcile DSL v1.
2//!
3//! The DSL implements the observe → effective → materialize model described
4//! in the [Reconcile DSL guide](https://docs.rs/zk-lsp/latest/zk_lsp/reconcile_dsl/).
5//!
6//! At evaluation time, the engine:
7//! 1. Builds a workspace snapshot (notes, checkboxes, typed metadata)
8//! 2. Loads and merges DSL modules from disk
9//! 3. Type-checks the merged module against the builtin surface
10//! 4. Evaluates `effective_checked` / `effective_meta` in topological order
11//! 5. Writes back fields declared by `materialized_fields`
12

20/// Stable identifier for a checklist item: note ID plus 0-based line index.

35/// Note / checklist status values used throughout the DSL.
36///
37/// Maps to the `checklist-status` TOML field and to the `Status` DSL type.
38#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
39pub enum Status {
40    /// No checklist — status not applicable.
41    None,
42    /// At least one item incomplete, none in progress.
43    Todo,
44    /// Mixed: some done, some not.
45    Wip,
46    /// All leaf items complete.
47    Done,
48}
49
50impl Status {
51    #[allow(dead_code)]
52    pub fn is_done(&self) -> bool {
53        matches!(self, Self::Done)
54    }
55
56    pub fn to_str(&self) -> &'static str {
57        match self {
58            Self::None => "none",
59            Self::Todo => "todo",
60            Self::Wip => "wip",
61            Self::Done => "done",
62        }
63    }
64
65    pub fn from_str(value: &str) -> Option<Self> {
66        match value {
67            "none" => Some(Self::None),
68            "todo" => Some(Self::Todo),
69            "wip" => Some(Self::Wip),
70            "done" => Some(Self::Done),
71            _ => None,
72        }
73    }
74
75    pub fn aggregate(statuses: &[Self]) -> Self {
76        let applicable: Vec<Self> = statuses
77            .iter()
78            .copied()
79            .filter(|status| *status != Self::None)
80            .collect();
81
82        if applicable.is_empty() {
83            return Self::None;
84        }
85        if applicable.iter().all(|status| *status == Self::Done) {
86            return Self::Done;
87        }
88        if applicable.iter().all(|status| *status == Self::Todo) {
89            return Self::Todo;
90        }
91        Self::Wip
92    }
93}
94
95impl fmt::Display for Status {
96    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
97        write!(f, "{}", self.to_str())
98    }
99}

101/// Checkbox writeback directives used during the materialize phase.
102///
103/// These control how a checkbox line should be written back to disk after
104/// `effective_checked` has produced a semantic `Status`.
105#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
106pub enum CheckboxWriteback {
107    /// Leave the source checkbox text unchanged.
108    Keep,
109    /// Materialize as `- [ ]`.
110    Unchecked,
111    /// Materialize as `- [x]`.
112    Checked,
113}
114
115impl CheckboxWriteback {
116    pub fn to_str(&self) -> &'static str {
117        match self {
118            Self::Keep => "keep",
119            Self::Unchecked => "unchecked",
120            Self::Checked => "checked",
121        }
122    }
123
124    pub fn from_str(value: &str) -> Option<Self> {
125        match value {
126            "keep" => Some(Self::Keep),
127            "unchecked" => Some(Self::Unchecked),
128            "checked" => Some(Self::Checked),
129            _ => None,
130        }
131    }
132}
133
134impl fmt::Display for CheckboxWriteback {
135    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
136        write!(f, "{}", self.to_str())
137    }
138}

140/// Runtime value in the Reconcile DSL evaluator.
141///
142/// The type system is checked statically before evaluation; these variants
143/// are the concrete runtime representations.  Users never interact with
144/// `Value` directly — it is the evaluator's internal currency.
145#[derive(Debug, Clone, PartialEq, Eq, Hash)]
146pub enum Value {
147    /// Logical `true` / `false`.
148    Bool(bool),
149    /// 64-bit signed integer.
150    Int(i64),
151    /// Absence value; returned by `parent` for root checklist items.
152    Nil,
153    /// One of `none` / `todo` / `wip` / `done`.
154    Status(Status),
155    /// One of `keep` / `unchecked` / `checked`.
156    CheckboxWriteback(CheckboxWriteback),
157    /// Homogeneous list; shared via reference-counting.
158    List(Rc<Vec<Value>>),
159    /// Runtime handle to a note (evaluator-internal).
160    NoteRef(NoteId),
161    /// Runtime handle to a checklist item (evaluator-internal).
162    CheckboxRef(CheckboxId),
163    /// String value returned by `observe_meta` for non-Status metadata fields.
164    String(Rc<str>),
165}
166
167#[derive(Debug, Clone, PartialEq, Eq, Hash)]
168pub enum Type {
169    Any,
170    Bool,
171    Int,
172    Nil,
173    Status,
174    CheckboxWriteback,
175    String,
176    NoteRef,
177    CheckboxRef,
178    List(Box<Type>),
179}
180
181impl fmt::Display for Type {
182    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
183        match self {
184            Type::Any => write!(f, "Any"),
185            Type::Bool => write!(f, "Bool"),
186            Type::Int => write!(f, "Int"),
187            Type::Nil => write!(f, "Nil"),
188            Type::Status => write!(f, "Status"),
189            Type::CheckboxWriteback => write!(f, "CheckboxWriteback"),
190            Type::String => write!(f, "String"),
191            Type::NoteRef => write!(f, "NoteRef"),
192            Type::CheckboxRef => write!(f, "CheckboxRef"),
193            Type::List(inner) => write!(f, "List({inner})"),
194        }
195    }
196}

202#[derive(Debug, Clone, PartialEq)]
203pub enum ParseError {
204    UnexpectedEof,
205    UnexpectedToken { got: String, expected: String },
206    InvalidExprHead(String),
207    DuplicateRule(String),
208    InvalidPolicyKey(String),
209    InvalidPolicyValue { key: String, value: String },
210}
211
212impl fmt::Display for ParseError {
213    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
214        match self {
215            ParseError::UnexpectedEof => write!(f, "unexpected end of input"),
216            ParseError::UnexpectedToken { got, expected } => {
217                write!(f, "unexpected token '{got}', expected {expected}")
218            }
219            ParseError::InvalidExprHead(s) => write!(f, "invalid expression head '{s}'"),
220            ParseError::DuplicateRule(s) => write!(f, "duplicate rule name '{s}'"),
221            ParseError::InvalidPolicyKey(k) => write!(f, "unknown policy key '{k}'"),
222            ParseError::InvalidPolicyValue { key, value } => {
223                write!(f, "invalid value '{value}' for policy key '{key}'")
224            }
225        }
226    }
227}

229#[derive(Debug, Clone, PartialEq)]
230pub enum TypeError {
231    UnknownVariable(String),
232    UnknownFunction(String),
233    TypeMismatch {
234        expected: Type,
235        got: Type,
236    },
237    IfBranchMismatch {
238        then_type: Type,
239        else_type: Type,
240    },
241    WrongArgCount {
242        name: String,
243        expected: usize,
244        got: usize,
245    },
246    UnsupportedHigherOrderArg {
247        name: String,
248    },
249}
250
251impl fmt::Display for TypeError {
252    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
253        match self {
254            TypeError::UnknownVariable(v) => write!(f, "unknown variable '{v}'"),
255            TypeError::UnknownFunction(n) => write!(f, "unknown function '{n}'"),
256            TypeError::TypeMismatch { expected, got } => {
257                write!(f, "type mismatch: expected {expected}, got {got}")
258            }
259            TypeError::IfBranchMismatch {
260                then_type,
261                else_type,
262            } => {
263                write!(
264                    f,
265                    "if branch type mismatch: then={then_type}, else={else_type}"
266                )
267            }
268            TypeError::WrongArgCount {
269                name,
270                expected,
271                got,
272            } => {
273                write!(f, "'{name}' expected {expected} args, got {got}")
274            }
275            TypeError::UnsupportedHigherOrderArg { name } => {
276                write!(f, "unsupported higher-order argument in '{name}'")
277            }
278        }
279    }
280}
