1/// BFS context export for AI consumption.
2///
3/// `export_context` starts at a given note ID, traverses outgoing links (or
4/// inbound links when `inverse = true`) up to `depth` hops, and returns a
5/// structured Markdown document.
6use std::collections::{HashMap, HashSet, VecDeque};
7
8use crate::config::WikiConfig;
9use crate::metadata;
10use crate::note_info;
11use crate::parser::{self, ChecklistStatus, Relation};
12
13/// Export a BFS context document starting from `entry_id` to the given `depth`.
14///
15/// When `inverse = true`, traverses inbound links (backlinks) instead of
16/// outgoing links, and reverses the output so ancestors appear first with the
17/// entry note last — "先给前提，再给结论".
18///
19/// Returns the Markdown string or an error if the entry note cannot be read.
20pub async fn export_context(
21    entry_id: &str,
22    depth: usize,
23    inverse: bool,
24    simple: bool,
25    config: &WikiConfig,
26) -> anyhow::Result<String> {
27    let reverse_map = build_reverse_map(config).await;
28    let metadata_snapshot = metadata::MetadataSnapshot::load(config)
29        .await
30        .unwrap_or_else(metadata::MetadataSnapshot::unavailable);
31
32    let mut queue: VecDeque<(String, usize)> = VecDeque::new();
33    let mut visited: HashSet<String> = HashSet::new();
34
35    queue.push_back((entry_id.to_string(), 0));
36    visited.insert(entry_id.to_string());
37
38    let mut sections: Vec<NoteSection> = Vec::new();
39
40    while let Some((id, d)) = queue.pop_front() {
41        let path = config.note_dir.join(format!("{id}.typ"));
42        let content = match tokio::fs::read_to_string(&path).await {
43            Ok(c) => c,
44            Err(_) => continue,
45        };
46
47        let header = parser::parse_header(&content).filter(|header| header.id == id);
48        let metadata_record = metadata_snapshot.record(&id).ok();
49        let title = header.as_ref().map(|h| h.title.clone()).unwrap_or_default();
50        let abstract_text = metadata_record
51            .as_ref()
52            .and_then(|m| m.abstract_text.clone())
53            .unwrap_or_default();
54        let keywords = metadata_record
55            .as_ref()
56            .map(|m| m.keywords.clone())
57            .unwrap_or_default();
58        let checklist_status = metadata_record
59            .as_ref()
60            .map(|m| m.checklist_status.clone())
61            .unwrap_or(ChecklistStatus::None);
62        let relation = metadata_record
63            .as_ref()
64            .map(|m| m.relation.clone())
65            .unwrap_or(Relation::Active);
66
67        // Extract outgoing refs (filtered: skips TOML block, comments, fences)
68        let out_refs: Vec<String> = parser::find_all_link_ids_filtered(&content)
69            .into_iter()
70            .collect::<HashSet<_>>()
71            .into_iter()
72            .collect::<Vec<_>>();
73        let inbound_refs = reverse_map.get(&id).cloned().unwrap_or_default();
74
75        // Enqueue unvisited neighbours if within depth
76        if d < depth {
77            let neighbours: &[String] = if inverse { &inbound_refs } else { &out_refs };
78            for ref_id in neighbours {
79                if visited.insert(ref_id.clone()) {
80                    queue.push_back((ref_id.clone(), d + 1));
81                }
82            }
83        }
84
85        // Extract body: lines from the title heading onwards
86        let body = extract_body(&content, &id);
87
88        let mut sorted_refs = out_refs.clone();
89        sorted_refs.sort();
90        let note_info = header
91            .as_ref()
92            .zip(metadata_record.as_ref())
93            .map(|(h, record)| {
94                note_info::build_note_info_value(
95                    &id,
96                    &path,
97                    h,
98                    record,
99                    &config.zk_config.metadata.fields,
100                )
101            });
102
103        sections.push(NoteSection {
104            id: id.clone(),
105            title,
106            abstract_text,
107            keywords,
108            checklist_status,
109            relation,
110            out_refs: sorted_refs,
111            inbound_refs,
112            body,
113            note_info,
114        });
115    }
116
117    // In inverse mode, reverse so ancestors appear first and entry note is last
118    if inverse {
119        sections.reverse();
120    }
121
122    let entry_title = if inverse {
123        sections.last().map(|s| s.title.as_str()).unwrap_or("")
124    } else {
125        sections.first().map(|s| s.title.as_str()).unwrap_or("")
126    };
127    let today = chrono_today();
128    let section_ids: HashSet<String> = sections.iter().map(|s| s.id.clone()).collect();
129
130    if simple {
131        let items = sections
132            .iter()
133            .filter_map(|s| s.note_info.clone())
134            .collect::<Vec<_>>();
135        return Ok(serde_json::to_string_pretty(&items)?);
136    }
137
138    let mut out = String::new();
139    if inverse {
140        out.push_str("# Inverse Context Export\n\n");
141    } else {
142        out.push_str("# ZK Context Export\n\n");
143    }
144    out.push_str(&format!("**Entry:** {entry_id} — {entry_title}\n"));
145    out.push_str(&format!("**Depth:** {depth}\n"));
146    out.push_str(&format!("**Generated:** {today}\n\n"));
147    out.push_str("---\n\n");
148
149    for section in &sections {
150        out.push_str(&format!("## {} · {}\n\n", section.id, section.title));
151
152        if !section.abstract_text.is_empty() {
153            out.push_str(&format!("> {}\n\n", section.abstract_text));
154        }
155
156        if !section.keywords.is_empty() {
157            out.push_str(&format!("**Keywords:** {}\n", section.keywords.join(", ")));
158        }
159
160        let cs = section.checklist_status.as_str();
161        let rel = match section.relation {
162            Relation::Active => "active",
163            Relation::Archived => "archived",
164            Relation::Legacy => "legacy",
165        };
166        out.push_str(&format!("**Status:** checklist={cs} · relation={rel}\n"));
167
168        if !section.out_refs.is_empty() {
169            let refs_str = section
170                .out_refs
171                .iter()
172                .map(|r| format!("@{r}"))
173                .collect::<Vec<_>>()
174                .join(", ");
175            out.push_str(&format!("**Outgoing links:** {refs_str}\n"));
176        }
177
178        let mut local_inbound = section
179            .inbound_refs
180            .iter()
181            .filter(|id| section_ids.contains(*id))
182            .cloned()
183            .collect::<Vec<_>>();
184        local_inbound.sort();
185        local_inbound.dedup();
186        if !local_inbound.is_empty() {
187            let refs_str = local_inbound
188                .iter()
189                .map(|r| format!("@{r}"))
190                .collect::<Vec<_>>()
191                .join(", ");
192            out.push_str(&format!("**Referenced by:** {refs_str}\n"));
193        }
194
195        out.push('\n');
196
197        if !section.body.is_empty() {
198            out.push_str(&section.body);
199            out.push('\n');
200        }
201
202        out.push_str("---\n\n");
203    }
204
205    Ok(out)
206}

208struct NoteSection {
209    id: String,
210    title: String,
211    abstract_text: String,
212    keywords: Vec<String>,
213    checklist_status: ChecklistStatus,
214    relation: Relation,
215    out_refs: Vec<String>,
216    inbound_refs: Vec<String>,
217    body: String,
218    note_info: Option<serde_json::Value>,
219}

221/// Build a reverse map: target_id → Vec<source_id> by scanning all .typ files.
222async fn build_reverse_map(config: &WikiConfig) -> HashMap<String, Vec<String>> {
223    let mut map: HashMap<String, Vec<String>> = HashMap::new();
224    let mut dir = match tokio::fs::read_dir(&config.note_dir).await {
225        Ok(d) => d,
226        Err(_) => return map,
227    };
228    while let Ok(Some(entry)) = dir.next_entry().await {
229        let path = entry.path();
230        let Some(ext) = path.extension() else {
231            continue;
232        };
233        if ext != "typ" {
234            continue;
235        }
236        let Some(stem) = path.file_stem().and_then(|s| s.to_str()) else {
237            continue;
238        };
239        if stem.len() != 10 || !stem.chars().all(|c| c.is_ascii_digit()) {
240            continue;
241        }
242        let source_id = stem.to_string();
243        let content = match tokio::fs::read_to_string(&path).await {
244            Ok(c) => c,
245            Err(_) => continue,
246        };
247        for id in parser::find_all_link_ids_filtered(&content) {
248            map.entry(id).or_default().push(source_id.clone());
249        }
250    }
251    map
252}

254/// Extract the body of a note: everything from the title line (`= ... <id>`) onwards.
255fn extract_body(content: &str, id: &str) -> String {
256    let needle = format!("<{id}>");
257    let lines: Vec<&str> = content.lines().collect();
258    let start = lines
259        .iter()
260        .position(|l| l.contains(&needle))
261        .unwrap_or(lines.len());
262    lines[start..].join("\n")
263}

265fn chrono_today() -> String {
266    // Use std::time to get a simple date string without a heavy dependency
267    // Format: YYYY-MM-DD approximated from UNIX time
268    use std::time::{SystemTime, UNIX_EPOCH};
269    let secs = SystemTime::now()
270        .duration_since(UNIX_EPOCH)
271        .map(|d| d.as_secs())
272        .unwrap_or(0);
273    // Simple conversion: days since epoch
274    let days = secs / 86400;
275    // Zeller-style calculation
276    let (y, m, d) = days_to_ymd(days);
277    format!("{y:04}-{m:02}-{d:02}")
278}
279
280/// Convert days since Unix epoch (1970-01-01) to (year, month, day).
281fn days_to_ymd(days: u64) -> (u32, u32, u32) {
282    // Algorithm from https://howardhinnant.github.io/date_algorithms.html
283    let z = days as i64 + 719468;
284    let era = if z >= 0 { z } else { z - 146096 } / 146097;
285    let doe = z - era * 146097;
286    let yoe = (doe - doe / 1460 + doe / 36524 - doe / 146096) / 365;
287    let y = yoe + era * 400;
288    let doy = doe - (365 * yoe + yoe / 4 - yoe / 100);
289    let mp = (5 * doy + 2) / 153;
290    let d = doy - (153 * mp + 2) / 5 + 1;
291    let m = if mp < 10 { mp + 3 } else { mp - 9 };
292    let y = if m <= 2 { y + 1 } else { y };
293    (y as u32, m as u32, d as u32)
294}

306    fn make_note_content(id: &str, title: &str, refs: &[&str]) -> String {
307        let refs_body: String = refs.iter().map(|r| format!("- [ ] @{r}\n")).collect();
308        format!(
309            "#import \"../include.typ\": *\n\
310             #let zk-metadata = zk_metadata(\"{id}\")\n\
311             #show: zettel.with(metadata: zk-metadata)\n\
312             \n\
313             = {title} <{id}>\n\
314             {refs_body}"
315        )
316    }
317
318    fn write_note(dir: &Path, id: &str, title: &str, refs: &[&str]) {
319        let content = make_note_content(id, title, refs);
320        std::fs::write(dir.join(format!("{id}.typ")), content).unwrap();
321        write_metadata_record(dir.parent().unwrap(), id);
322    }
323
324    fn write_metadata_record(root: &Path, id: &str) {
325        let path = root.join("metadata.toml");
326        if !path.exists() {
327            std::fs::write(&path, "format-version = 1\n\n").unwrap();
328        }
329        let mut content = std::fs::read_to_string(&path).unwrap();
330        content.push_str(&format!(
331            "[notes.\"{id}\"]\n\
332             schema-version = 1\n\
333             aliases = []\n\
334             abstract = \"\"\n\
335             keywords = []\n\
336             generated = true\n\
337             checklist-status = \"none\"\n\
338             relation = \"active\"\n\
339             relation-target = []\n\n"
340        ));
341        std::fs::write(path, content).unwrap();
342    }
