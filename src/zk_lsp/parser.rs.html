87/// Scan `content` for a canonical central metadata binding:
88/// `#let zk-metadata = zk_metadata("ID")`.
89pub fn find_metadata_binding(content: &str) -> Option<MetadataBinding> {
90    let mut found = None;
91    for (line_idx, line) in content.lines().enumerate() {
92        let Some(caps) = RE_BINDING.captures(line) else {
93            continue;
94        };
95        if found.is_some() {
96            return None;
97        }
98        found = Some(MetadataBinding {
99            line_idx,
100            id: caps.get(1)?.as_str().to_string(),
101        });
102    }
103    found
104}

106/// Parse the structural header of a central-metadata note.
107pub fn parse_header(content: &str) -> Option<NoteHeader> {
108    let mut binding = None;
109    let mut title = None;
110
111    for (line_idx, line) in content.lines().enumerate() {
112        if let Some(captures) = RE_BINDING.captures(line) {
113            if binding.is_some() {
114                return None;
115            }
116            binding = Some(MetadataBinding {
117                line_idx,
118                id: captures.get(1)?.as_str().to_string(),
119            });
120        }
121
122        if title.is_none() {
123            if let Some(captures) = RE_TITLE.captures(line) {
124                let matched = captures.get(0)?.as_str();
125                title = Some(NoteHeader {
126                    id: captures.get(1)?.as_str().to_string(),
127                    title: matched
128                        .trim_start_matches('=')
129                        .trim()
130                        .rsplit_once('<')
131                        .map(|(title, _)| title.trim().to_string())
132                        .unwrap_or_default(),
133                    title_line_idx: line_idx,
134                });
135            }
136        }
137    }
138
139    let binding = binding?;
140    let title = title?;
141    if binding.line_idx >= title.title_line_idx || binding.id != title.id {
142        return None;
143    }
144    Some(title)
145}

174/// Parse all checklist items from `content`, skipping fenced code blocks.
175/// Items with `@(\d{10})` in their text become `Ref` items; all others are `Local`.
176/// `RefTarget.byte_start`/`byte_end` are byte offsets of `@ID` within the full line.
177pub fn parse_checklist_items(content: &str) -> Vec<ChecklistItem> {
178    let mut items = Vec::new();
179    let mut in_fence = false;
180
181    for (line_idx, line) in content.lines().enumerate() {
182        let trimmed = line.trim_start();
183        if trimmed.starts_with("```") {
184            in_fence = !in_fence;
185            continue;
186        }
187        if in_fence {
188            continue;
189        }
190        if !(trimmed.starts_with("- [") && trimmed.len() >= 5) {
191            continue;
192        }
193        let marker = trimmed.chars().nth(3).unwrap_or(' ');
194        if marker != 'x' && marker != 'X' && marker != ' ' {
195            continue;
196        }
197        let checked = marker == 'x' || marker == 'X';
198        let indent = line.len() - trimmed.len();
199        // prefix_len: bytes before the checklist body (indent + "- [x] ")
200        let prefix_len = indent + 6;
201        // text after `- [x] ` (or `- [ ] `)
202        let body = trimmed.get(6..).unwrap_or("");
203        let text = body.to_string();
204        let targets: Vec<RefTarget> = RE_ID_REF
205            .captures_iter(body)
206            .filter_map(|captures| {
207                let full = captures.get(0)?;
208                if is_followed_by_ascii_digit(body, full.end()) {
209                    return None;
210                }
211                Some(RefTarget {
212                    target_id: captures.get(1)?.as_str().to_string(),
213                    byte_start: (prefix_len + full.start()) as u32,
214                    byte_end: (prefix_len + full.end()) as u32,
215                })
216            })
217            .collect();
218        let kind = if targets.is_empty() {
219            ChecklistItemKind::Local
220        } else {
221            ChecklistItemKind::Ref { targets }
222        };
223        items.push(ChecklistItem {
224            checked,
225            kind,
226            text,
227            line_idx,
228            indent,
229        });
230    }
231    items
232}

234/// Evaluate the semantic truth of a single checklist item.
235/// `Local` items: truth = checkbox state.
236/// `Ref` items: truth = `∀ t ∈ targets: done_lookup(t.target_id)` — never the rendered checkbox.
237pub fn eval_item_truth(item: &ChecklistItem, done_lookup: &impl Fn(&str) -> bool) -> bool {
238    match &item.kind {
239        ChecklistItemKind::Local => item.checked,
240        ChecklistItemKind::Ref { targets } => targets.iter().all(|t| done_lookup(&t.target_id)),
241    }
242}

244fn is_leaf(items: &[ChecklistItem], idx: usize) -> bool {
245    idx + 1 >= items.len() || items[idx + 1].indent <= items[idx].indent
246}

248/// Compute whether a note is done based on its checklist items and a dependency lookup.
249///
250/// Only **leaf items** participate: a leaf is an item with no subsequent item
251/// with strictly greater indent before the next same-or-lesser-indent item.
252/// Non-leaf LocalItems are derived display views and must not be counted as source facts.
253/// If there are no items, returns `false` (caller should check metadata separately).
254pub fn compute_note_done_from_items(
255    items: &[ChecklistItem],
256    done_lookup: &impl Fn(&str) -> bool,
257) -> bool {
258    let leaves: Vec<&ChecklistItem> = items
259        .iter()
260        .enumerate()
261        .filter(|(i, _)| is_leaf(items, *i))
262        .map(|(_, item)| item)
263        .collect();
264    if leaves.is_empty() {
265        return false;
266    }
267    leaves.iter().all(|item| eval_item_truth(item, done_lookup))
268}

270/// Convert a byte offset within `s` to a UTF-16 code-unit offset.
271/// LSP `character` positions are UTF-16 code units, not bytes or scalar values.
272pub fn byte_to_utf16(s: &str, byte_offset: usize) -> u32 {
273    s[..byte_offset].chars().map(|c| c.len_utf16() as u32).sum()
274}
275
276fn utf16_to_byte(s: &str, character: u32) -> Option<usize> {
277    if character == 0 {
278        return Some(0);
279    }
280
281    let mut utf16_units = 0;
282    for (byte_idx, ch) in s.char_indices() {
283        utf16_units += ch.len_utf16() as u32;
284        if utf16_units == character {
285            return Some(byte_idx + ch.len_utf8());
286        }
287        if utf16_units > character {
288            return None;
289        }
290    }
291
292    (utf16_units == character).then_some(s.len())
293}

295fn is_followed_by_ascii_digit(text: &str, byte_offset: usize) -> bool {
296    text.as_bytes()
297        .get(byte_offset)
298        .is_some_and(|byte| byte.is_ascii_digit())
299}

301fn push_refs_from_segment(
302    refs: &mut Vec<RefOccurrence>,
303    line: u32,
304    segment: &str,
305    segment_start: usize,
306) {
307    for captures in RE_ID_REF.captures_iter(segment) {
308        let Some(full) = captures.get(0) else {
309            continue;
310        };
311        if is_followed_by_ascii_digit(segment, full.end()) {
312            continue;
313        }
314        let Some(id) = captures.get(1) else {
315            continue;
316        };
317        refs.push(RefOccurrence {
318            id: id.as_str().to_string(),
319            line,
320            start_char: (segment_start + full.start()) as u32,
321            end_char: (segment_start + full.end()) as u32,
322        });
323    }
324}

326fn visible_segments(content: &str) -> Vec<VisibleSegment> {
327    let mut segments = Vec::new();
328    let mut in_block_comment = false;
329    let mut in_fence = false;
330
331    for (line_idx, line) in content.lines().enumerate() {
332        if !in_block_comment && line.trim_start().starts_with("```") {
333            in_fence = !in_fence;
334            continue;
335        }
336        if in_fence {
337            continue;
338        }
339
340        let mut pos = 0;
341        while pos < line.len() {
342            if in_block_comment {
343                let Some(end_offset) = line[pos..].find("*/") else {
344                    break;
345                };
346                pos += end_offset + 2;
347                in_block_comment = false;
348                continue;
349            }
350
351            if let Some(start_offset) = line[pos..].find("/*") {
352                let comment_start = pos + start_offset;
353                if pos < comment_start {
354                    segments.push(VisibleSegment {
355                        line: line_idx as u32,
356                        start: pos,
357                        end: comment_start,
358                    });
359                }
360                pos = comment_start + 2;
361                in_block_comment = true;
362            } else {
363                segments.push(VisibleSegment {
364                    line: line_idx as u32,
365                    start: pos,
366                    end: line.len(),
367                });
368                break;
369            }
370        }
371    }
372
373    segments
374}

376fn range_contains_position(range: Range, position: Position) -> bool {
377    position.line == range.start.line
378        && position.character >= range.start.character
379        && position.character <= range.end.character
380}

382fn line_byte_range_to_lsp(line_idx: u32, line: &str, start: usize, end: usize) -> Range {
383    Range {
384        start: Position {
385            line: line_idx,
386            character: byte_to_utf16(line, start),
387        },
388        end: Position {
389            line: line_idx,
390            character: byte_to_utf16(line, end),
391        },
392    }
393}

395/// Find all exact `@ID` occurrences in content, where `ID` is ten ASCII digits.
396/// `start_char` / `end_char` are **byte** offsets within the line (not UTF-16).
397/// Convert with `byte_to_utf16` before using as LSP character positions.
398pub fn find_all_refs(content: &str) -> Vec<RefOccurrence> {
399    let mut refs = Vec::new();
400    for (line_num, line) in content.lines().enumerate() {
401        push_refs_from_segment(&mut refs, line_num as u32, line, 0);
402    }
403    refs
404}

406/// Find all exact `@ID` occurrences in visible content, skipping block comments
407/// and fenced code blocks.
408pub fn find_all_refs_filtered(content: &str) -> Vec<RefOccurrence> {
409    let lines: Vec<&str> = content.lines().collect();
410    let mut refs = Vec::new();
411
412    for visible in visible_segments(content) {
413        let Some(line) = lines.get(visible.line as usize) else {
414            continue;
415        };
416        push_refs_from_segment(
417            &mut refs,
418            visible.line,
419            &line[visible.start..visible.end],
420            visible.start,
421        );
422    }
423
424    refs
425}

427/// Return the complete visible `@ID` token at an LSP position.
428pub fn note_ref_at_position(content: &str, position: Position) -> Option<NoteRefAtPosition> {
429    let lines: Vec<&str> = content.lines().collect();
430    find_all_refs_filtered(content)
431        .into_iter()
432        .filter(|occurrence| occurrence.line == position.line)
433        .find_map(|occurrence| {
434            let line = lines.get(occurrence.line as usize)?;
435            let range = line_byte_range_to_lsp(
436                occurrence.line,
437                line,
438                occurrence.start_char as usize,
439                occurrence.end_char as usize,
440            );
441            range_contains_position(range, position).then_some(NoteRefAtPosition {
442                id: occurrence.id,
443                range,
444            })
445        })
446}

448/// Return the zero-width insertion range after a visible bare `@` trigger.
449pub fn note_ref_completion_range(content: &str, position: Position) -> Option<Range> {
450    let line = content.lines().nth(position.line as usize)?;
451    let cursor_byte = utf16_to_byte(line, position.character)?;
452    let (at_byte, previous) = line[..cursor_byte].char_indices().next_back()?;
453    if previous != '@' {
454        return None;
455    }
456
457    let visible = visible_segments(content).into_iter().any(|segment| {
458        segment.line == position.line && at_byte >= segment.start && at_byte < segment.end
459    });
460    visible.then_some(Range {
461        start: position,
462        end: position,
463    })
464}

466/// Return the current note's identity when the position is on its canonical
467/// metadata binding or title-heading ID.
468pub fn note_identity_at_position(content: &str, position: Position) -> Option<String> {
469    let header = parse_header(content)?;
470    let binding = find_metadata_binding(content)?;
471    let lines: Vec<&str> = content.lines().collect();
472
473    let (line_idx, token, use_last_match) = if position.line == header.title_line_idx as u32 {
474        (header.title_line_idx, format!("<{}>", header.id), true)
475    } else if position.line == binding.line_idx as u32 {
476        (binding.line_idx, format!("\"{}\"", header.id), false)
477    } else {
478        return None;
479    };
480
481    let line = *lines.get(line_idx)?;
482    let start = if use_last_match {
483        line.rfind(&token)?
484    } else {
485        line.find(&token)?
486    };
487    let range = line_byte_range_to_lsp(position.line, line, start, start + token.len());
488    range_contains_position(range, position).then_some(header.id)
489}

491/// Find all link target IDs in content, combining both `@ID` and `<ID>` forms,
492/// while skipping block comments and fenced code blocks.
493///
494/// The note's own title ID (`= Title <ID>`) is excluded to avoid treating the
495/// note header as a self-link.
496pub fn find_all_link_ids_filtered(content: &str) -> Vec<String> {
497    let self_id = parse_header(content).map(|h| h.id);
498    let mut ids: Vec<String> = find_all_refs_filtered(content)
499        .into_iter()
500        .map(|r| r.id)
501        .collect();
502
503    let mut in_block_comment = false;
504    let mut in_fence = false;
505
506    for line in content.lines() {
507        let mut visible_segments = Vec::new();
508        let mut pos = 0usize;
509        loop {
510            if in_block_comment {
511                let Some(end_offset) = line[pos..].find("*/") else {
512                    break;
513                };
514                in_block_comment = false;
515                pos += end_offset + 2;
516                continue;
517            }
518
519            let trimmed = line[pos..].trim_start();
520            if pos == 0 && trimmed.starts_with("```") {
521                in_fence = !in_fence;
522                break;
523            }
524            if in_fence {
525                break;
526            }
527
528            let remaining = &line[pos..];
529            let bc_start = remaining.find("/*");
530            let fence_start = remaining.find("```");
531            let next_cut = match (bc_start, fence_start) {
532                (Some(bc), Some(fence)) => Some(bc.min(fence)),
533                (Some(bc), None) => Some(bc),
534                (None, Some(fence)) => Some(fence),
535                (None, None) => None,
536            };
537
538            match next_cut {
539                Some(offset) => {
540                    visible_segments.push((pos, pos + offset));
541                    let abs = pos + offset;
542                    if remaining[offset..].starts_with("/*") {
543                        in_block_comment = true;
544                        pos = abs + 2;
545                    } else {
546                        in_fence = true;
547                        break;
548                    }
549                }
550                None => {
551                    visible_segments.push((pos, line.len()));
552                    break;
553                }
554            }
555        }
556
557        for (seg_start, seg_end) in visible_segments {
558            let segment = &line[seg_start..seg_end];
559            for cap in RE_ANGLE_ID.captures_iter(segment) {
560                let id = cap.get(1).unwrap().as_str();
561                if self_id.as_deref() == Some(id) {
562                    continue;
563                }
564                ids.push(id.to_string());
565            }
566        }
567    }
568
569    ids
570}

572/// A heading parsed from note content (outside fenced code).
573#[derive(Debug, Clone)]
574#[allow(dead_code)]
575pub struct Heading {
576    pub level: u32,
577    pub text: String,
578    pub line_idx: usize,
579}
580
581/// Parse all headings from `content`, skipping fenced code blocks.
582/// For the title heading (matching `= Title <YYMMDDHHMM>`), the ` <ID>` suffix is stripped.
583#[allow(dead_code)]
584pub fn parse_headings(content: &str) -> Vec<Heading> {
585    static RE_HEADING: Lazy<Regex> = Lazy::new(|| Regex::new(r"^(=+)\s+(.+)").unwrap());
586    static RE_ID_SUFFIX: Lazy<Regex> = Lazy::new(|| Regex::new(r"\s+<[0-9]{10}>$").unwrap());
587
588    let mut headings = Vec::new();
589    let mut in_fence = false;
590
591    for (line_idx, line) in content.lines().enumerate() {
592        if line.trim_start().starts_with("```") {
593            in_fence = !in_fence;
594            continue;
595        }
596        if in_fence {
597            continue;
598        }
599        if let Some(cap) = RE_HEADING.captures(line) {
600            let level = cap[1].len() as u32;
601            let raw_text = cap[2].trim().to_string();
602            let text = RE_ID_SUFFIX.replace(&raw_text, "").to_string();
603            headings.push(Heading {
604                level,
605                text,
606                line_idx,
607            });
608        }
609    }
610    headings
611}
